use focal_core::{Core, PendingState};
use focal_model::*;
use std::{collections::BTreeSet, path::Path};
fn ledger()->LedgerId { LedgerId{tenant:TenantId::from_u128(10),session:SessionId::from_u128(20)} }
fn input(n:u128, actor:u128, command:Command)->AuthenticatedInput {AuthenticatedInput{ledger:ledger(),principal:ParticipantId::from_u128(actor),request_epoch:RequestEpoch(1),request_id:RequestId::from_u128(n),expected_revision:None,authority:AuthorityContext{runtime:true,cause:Cause::Root(RootCommandId::from_u128(30)),policy_revision:1,logical_time:1000,evidence:vec![]},command}}
fn claim()->NewClaim {let cid=ClaimId::from_u128(100); let vid=ValidationId::from_u128(100100);let v=ValidationContent{ledger:ledger(),schema:1,claim:cid,kind:ValidationKind::Receipt,phase:ValidationPhase::WholeWork,mode:ValidationMode::Required,description:"acknowledged receipt".into(),quality_bar:None,evaluator:ParticipantId::from_u128(3),handlers:vec![],evidence_schemas:BTreeSet::new(),contributed_by:BTreeSet::from([ParticipantId::from_u128(1)]),policy_revision:1};NewClaim{id:cid,content:ClaimContent{ledger:ledger(),schema:1,occurrence:OccurrenceId::from_u128(100),description:"return evidence".into(),relations:BTreeSet::from([Relation{kind:RelationKind::Issuer,target:RelationTarget::Participant(ParticipantId::from_u128(1))},Relation{kind:RelationKind::Subject,target:RelationTarget::Participant(ParticipantId::from_u128(2))},Relation{kind:RelationKind::ClaimAction,target:RelationTarget::Action(ActionType::Work)},Relation{kind:RelationKind::CausedBy,target:RelationTarget::Root(RootCommandId::from_u128(30))}]),scopes:BTreeSet::new(),requirements:vec![RequirementRef{id:vid,specification:v.specification_hash().unwrap()}],deadline:None},validations:vec![NewValidation{id:vid,content:v}]}}
fn save(root:&Path,name:&str,bytes:Vec<u8>){std::fs::write(root.join(name),bytes).unwrap();}
fn main(){let root=Path::new("/private/tmp/focal-v1-golden");std::fs::create_dir_all(root).unwrap();let mut core=Core::new(ledger(),Limits::default());save(root,"empty.cp1",core.encode_checkpoint().unwrap());
for actor in 1..=3 {let p=core.prepare(&input(900+actor,actor,Command::NegotiateEpoch{epoch:RequestEpoch(1)})).unwrap();core.apply(SessionSeq(core.sequence().0+1),p).unwrap();}save(root,"base.cp1",core.encode_checkpoint().unwrap());
let cid=ClaimId::from_u128(100);let fence=ReceiptFence{receipt:ReceiptId::from_u128(101),epoch:1};let set=EvidenceSetId::from_u128(103);let tid=TestamentId::from_u128(104);
let artifact=NewArtifact{id:ArtifactId::from_u128(102),content:ArtifactContent{ledger:ledger(),schema:1,kind:"test-report".into(),schema_hash:ContentHash([7;32]),metadata:vec![],payload:ArtifactPayload::Inline(b"immutable proof".to_vec()),producer:ParticipantId::from_u128(2),receipt:Some(fence),inputs:BTreeSet::new(),visibility:BTreeSet::new()}};let aref=ArtifactRef{id:artifact.id,hash:artifact.content.content_hash().unwrap()};
let commands=vec![(1,Command::GenerateClaim{claim:claim()}),(1,Command::PostClaim{claim:cid}),(2,Command::AcquireReceipt{claim:cid,receipt:fence.receipt,epoch:1}),(2,Command::BeginEvidenceSet{claim:cid,receipt:fence,evidence_set:set}),(2,Command::AttachArtifact{claim:cid,receipt:fence,evidence_set:set,artifact}),(2,Command::CloseTestament{claim:cid,receipt:fence,testament:tid,evidence_set:set,manifest:vec![aref],summary:"finished with proof".into(),confidence:Confidence::Committed,outcome:OutcomeKind::Complete}),(1,Command::AcknowledgeTestament{claim:cid,testament:tid}),(1,Command::BeginWholeWorkValidation{claim:cid}),(1,Command::CompleteWholeWork{claim:cid})];
for (index,(actor,command)) in commands.into_iter().enumerate(){let mut i=input(1000+index as u128,actor,command);if let Command::AttachArtifact{artifact,..}=&i.command{i.authority.evidence.push(EvidenceAttestation{descriptor_hash:artifact.content.content_hash().unwrap(),custody_revision:1,durable:true,schema_valid:true});}
if index%2==0 {let p=core.prepare(&i).unwrap();let mut bytes=b"FOCALOP1".to_vec();bytes.extend(postcard::to_stdvec(&p).unwrap());save(root,&format!("{index:02}.entry"),bytes);let out=core.apply(SessionSeq(core.sequence().0+1),p).unwrap();save(root,&format!("{index:02}.result"),postcard::to_stdvec(&out).unwrap());}else{let m=ManagedAuthenticatedInput{key:ManagedRequestKey{stream:RequestStreamIdentity{cluster:[9;16],ledger:ledger(),principal:i.principal,slot:0,generation:1},ordinal:index as u64+1,id:i.request_id},expected_revision:i.expected_revision,authority:i.authority,command:i.command};let staged=core.stage_managed_pending_bounded(&PendingState::new(),&m,16*1024*1024).unwrap();let mut bytes=b"FOCALMD1".to_vec();bytes.extend(postcard::to_stdvec(staged.prepared()).unwrap());save(root,&format!("{index:02}.entry"),bytes);let out=core.publish_managed(staged).unwrap();save(root,&format!("{index:02}.result"),postcard::to_stdvec(&out).unwrap());}
if index==4{save(root,"open-evidence.cp1",core.encode_checkpoint().unwrap());}if index==5{save(root,"closed-evidence.cp1",core.encode_checkpoint().unwrap());}}
save(root,"final.cp1",core.encode_checkpoint().unwrap());}

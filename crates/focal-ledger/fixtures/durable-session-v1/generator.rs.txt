// Original native Session writer capture. This ignored test is an explicit
// one-off generator, never an ordinary golden-test regeneration path.
use std::io::Write;

struct Capture {
    root: std::path::PathBuf,
    files: Vec<(String, usize, String, String)>,
}
impl Capture {
    fn bytes(&mut self, name: &str, bytes: &[u8], shape: &str) {
        let mut file = std::fs::OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(self.root.join(name))
            .unwrap();
        file.write_all(bytes).unwrap();
        self.files.push((
            name.to_owned(),
            bytes.len(),
            blake3::hash(bytes).to_hex().to_string(),
            shape.to_owned(),
        ));
    }
    fn rows<T: Serialize>(&mut self, name: &str, values: &T, shape: &str) {
        self.bytes(name, &postcard::to_stdvec(values).unwrap(), shape);
    }
    fn envelope<T: Serialize>(&mut self, name: &str, magic: &[u8], value: &T) -> Vec<u8> {
        let mut bytes = magic.to_vec();
        bytes.extend(postcard::to_stdvec(value).unwrap());
        self.bytes(name, &bytes, std::any::type_name::<T>());
        bytes
    }
    fn finish(self) {
        let mut text = String::from("file\tbytes\tblake3\tshape\n");
        for (name, size, digest, shape) in self.files {
            text.push_str(&format!("{name}\t{size}\t{digest}\t{shape}\n"));
        }
        std::fs::write(self.root.join("capture-files.tsv"), text).unwrap();
        std::fs::write(
            self.root.join("capture-executable.txt"),
            std::env::current_exe().unwrap().display().to_string(),
        )
        .unwrap();
    }
}
fn native_rows<T: serde::de::DeserializeOwned>(relative: &str) -> T {
    let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
        .parent()
        .unwrap();
    postcard::from_bytes(&std::fs::read(root.join(relative)).unwrap()).unwrap()
}
fn native_copy<T: Serialize + serde::de::DeserializeOwned>(value: &T) -> T {
    postcard::from_bytes(&postcard::to_stdvec(value).unwrap()).unwrap()
}
fn capture_membership() -> (
    Vec<SessionMembershipRequest>,
    Vec<SessionMembershipReceipt>,
    Vec<MembershipContext>,
) {
    let ordinary = MembershipConfiguration {
        voters: vec![1, 2],
        learners: vec![3],
        ..Default::default()
    };
    let joint = MembershipConfiguration {
        voters: vec![1, 2],
        learners: vec![4],
        voters_outgoing: vec![1, 3],
        learners_next: vec![3],
        auto_leave: true,
    };
    let changes = [
        MembershipChange::AddLearner { node: 4 },
        MembershipChange::Promote { node: 3 },
        MembershipChange::Remove { node: 2 },
        MembershipChange::LeaveJoint,
    ];
    let mut requests = Vec::new();
    let mut receipts = Vec::new();
    let mut contexts = Vec::new();
    for (i, change) in changes.into_iter().enumerate() {
        let request = SessionMembershipRequest {
            id: u128::try_from(i + 1).unwrap().to_be_bytes(),
            expected_index: 127 + i as u64,
            expected: if change == MembershipChange::LeaveJoint {
                joint.clone()
            } else {
                ordinary.clone()
            },
            change,
        };
        let hash = request.hash().unwrap();
        let configuration = change.apply_to(&request.expected).unwrap();
        receipts.push(SessionMembershipReceipt {
            id: request.id,
            request_hash: hash,
            index: 255 + i as u64,
            term: 128 + i as u64,
            configuration,
        });
        contexts.push(MembershipContext {
            ledger: identity(),
            id: request.id,
            expected_index: request.expected_index,
            change,
            hash,
        });
        requests.push(request);
    }
    (requests, receipts, contexts)
}
fn capture_placements() -> Vec<StoredPlacement> {
    use focal_directory::{DurabilityIntent, FailureClass, Placement, PlacementPolicy, RegionId};
    let mut rows = Vec::new();
    for (i, kind) in [
        SessionFenceKind::Created,
        SessionFenceKind::Cutover,
        SessionFenceKind::Activated,
    ]
    .into_iter()
    .enumerate()
    {
        for (j, survive) in [FailureClass::Node, FailureClass::Zone, FailureClass::Region]
            .into_iter()
            .enumerate()
        {
            let populated = j != 0;
            let placement = PlacementSpec {
                policy: PlacementPolicy {
                    durability: DurabilityIntent {
                        survive,
                        max_failures: if populated { u16::MAX } else { 0 },
                    },
                    residency: if populated {
                        [RegionId::from_u128(1), RegionId::from_u128(u128::MAX)]
                            .into_iter()
                            .collect()
                    } else {
                        Default::default()
                    },
                    home_regions: if populated {
                        [RegionId::from_u128(2)].into_iter().collect()
                    } else {
                        Default::default()
                    },
                    required_memory: if populated { u64::MAX } else { 0 },
                },
                placement: Placement {
                    voters: [(1, 128), (2, u64::MAX)].into_iter().collect(),
                    materializers: [(1, 128), (2, u64::MAX), (3, 255)].into_iter().collect(),
                    content_copies: [(1, 128), (2, u64::MAX), (4, 256)].into_iter().collect(),
                    preferred_leader: 2,
                },
            };
            let request = SessionPlacementRequest {
                expected_index: if i == 0 { 0 } else { 128 },
                expected_configuration_index: 255,
                operation: OperationId::from_u128(100 + (i * 3 + j) as u128),
                kind,
                from_route: RouteEpoch(if i == 0 { 0 } else { 1 }),
                to_route: RouteEpoch(if i == 0 { 1 } else { 2 }),
                membership_epoch: if i == 0 { 1 } else { 128 },
                placement_epoch: if i == 0 { 1 } else { 2 },
                placement,
            };
            let record = PlacementRecord {
                schema: 1,
                ledger: identity(),
                group: LogGroupId::from_u128(2),
                genesis: ContentHash([71; 32]),
                sequence: SessionSeq(127),
                configuration: MembershipConfiguration {
                    voters: vec![1, 2],
                    learners: vec![3],
                    ..Default::default()
                },
                request,
            };
            let encoded = encode_placement(&record).unwrap();
            let fence = SessionFence {
                kind,
                ledger: identity(),
                log_group: record.group,
                operation: record.request.operation,
                sequence: record.sequence,
                index: RaftIndex(256 + i as u64),
                term: RaftTerm(129 + j as u64),
                from_route: record.request.from_route,
                to_route: record.request.to_route,
                membership_epoch: record.request.membership_epoch,
                placement_epoch: record.request.placement_epoch,
                placement_digest: focal_directory::placement_digest(&record.request.placement)
                    .unwrap(),
                record_hash: ContentHash(*blake3::hash(&encoded).as_bytes()),
            };
            rows.push(StoredPlacement { record, fence });
        }
    }
    rows
}
fn capture_base(
    populated: bool,
    checkpoints: &[CursorCheckpoint],
    receipts: &[CursorReceipt],
    deltas: &[Delta],
) -> SnapshotEnvelopeV2 {
    let core = if populated {
        include_bytes!("../../focal-core/fixtures/durable-v1-nested/broad.cp1").to_vec()
    } else {
        Core::new(identity(), Limits::default())
            .encode_checkpoint()
            .unwrap()
    };
    SnapshotEnvelopeV2 {
        schema: 2,
        ledger: identity(),
        raft_index: if populated { 777 } else { 1 },
        core,
        cursors: checkpoints[if populated { 2 } else { 0 }].clone(),
        cursor_meta: CursorMetadata {
            receipts: if populated {
                receipts.iter().cloned().map(|row| (row.key, row)).collect()
            } else {
                BTreeMap::new()
            },
            owners: if populated {
                receipts
                    .iter()
                    .filter_map(|row| {
                        row.record
                            .as_ref()
                            .map(|record| (record.token.key.consumer, row.key.principal))
                    })
                    .collect()
            } else {
                BTreeMap::new()
            },
        },
        delta_floor: SessionSeq(if populated { 127 } else { 0 }),
        deltas: if populated {
            deltas.to_vec()
        } else {
            Vec::new()
        },
    }
}

fn capture_live_snapshot(capture: &mut Capture, name: &str, session: &mut Session) -> Vec<u8> {
    let encoded = session.encode_checkpoint(false).unwrap().unwrap();
    capture.bytes(
        &format!("live-{name}.bin"),
        &encoded.bytes,
        "actual Session::encode_checkpoint writer output",
    );
    capture.rows(
        &format!("live-{name}.context"),
        &(
            session.applied_raft,
            session.status().term,
            session.consensus.membership_configuration(),
        ),
        "(u64 index,u64 term,MembershipConfiguration)",
    );
    capture.bytes(
        &format!("live-{name}.core"),
        &session.core.encode_checkpoint().unwrap(),
        "actual resulting Core checkpoint",
    );
    encoded.bytes
}
fn capture_real_history(capture: &mut Capture) {
    let directory = tempfile::tempdir().unwrap();
    let mut session = Session::open(
        directory.path(),
        identity(),
        config(),
        SessionLimits::default(),
    )
    .unwrap();
    elect(&mut session);
    let epoch_input = epoch(1);
    session.submit_local(&epoch_input).unwrap();
    let cursor = register(&session, 501, false);
    session.submit_cursor_local(&cursor).unwrap();
    capture.rows(
        "live-epoch-input.bin",
        &epoch_input,
        "actual admitted AuthenticatedInput",
    );
    capture.rows(
        "live-cursor-input.bin",
        &cursor,
        "actual admitted CursorInput",
    );
    let ss3 = capture_live_snapshot(capture, "ss3", &mut session);
    let decoded: SnapshotEnvelopeV3 =
        postcard::from_bytes(ss3.strip_prefix(SNAPSHOT_V3_MAGIC).unwrap()).unwrap();
    let ss1 = SnapshotEnvelope {
        schema: 1,
        ledger: decoded.state.ledger,
        raft_index: decoded.state.raft_index,
        core: decoded.state.core.clone(),
    };
    capture.envelope("live-ss1.bin", SNAPSHOT_MAGIC, &ss1);
    capture.envelope("live-ss2.bin", SNAPSHOT_V2_MAGIC, &decoded.state);
    for (kind, operation) in [
        (SessionFenceKind::Created, 601),
        (SessionFenceKind::Cutover, 602),
        (SessionFenceKind::Activated, 602),
    ] {
        let request = placement_request(&session, kind, operation);
        session.propose_placement(&request).unwrap();
        session.poll().unwrap();
        assert!(session.placement_receipt(&request).unwrap().is_some());
    }
    capture_live_snapshot(capture, "ss4", &mut session);
    let stream = stream_register(&mut session, 0, 4);
    let domain = managed_claim(stream, 1);
    let receipt = commit_managed(&mut session, &domain);
    capture.rows(
        "live-managed-input.bin",
        &domain,
        "actual admitted ManagedAuthenticatedInput",
    );
    capture.rows(
        "live-managed-receipt.bin",
        &receipt,
        "actual committed ManagedReceipt",
    );
    let old = ack(
        &session,
        701,
        Position::resolved(identity(), session.sequence()),
    );
    let cursor = ManagedCursorInput {
        key: ManagedRequestKey {
            stream,
            ordinal: 2,
            id: RequestId::from_u128(701),
        },
        intent_hash: old.intent_hash,
        command: old.command,
    };
    session.propose_managed_cursor(&cursor, false).unwrap();
    session.poll().unwrap();
    let revision = match session
        .request_streams
        .state(stream.principal, stream.slot)
        .unwrap()
    {
        RequestStreamState::Active { revision, .. } => revision,
        _ => panic!("active stream"),
    };
    let seal = stream_control(
        702,
        RequestStreamCommand::Seal {
            key: ManagedRequestKey {
                stream,
                ordinal: 3,
                id: RequestId::from_u128(703),
            },
            expected_revision: revision,
            family: ManagedRequestFamily::Domain,
            intent_hash: ContentHash([84; 32]),
        },
    );
    session.propose_request_stream(&seal).unwrap();
    session.poll().unwrap();
    assert!(session.request_stream_receipt(&seal).unwrap().is_some());
    capture.rows(
        "live-managed-cursor-input.bin",
        &cursor,
        "actual admitted ManagedCursorInput",
    );
    capture.rows(
        "live-seal-input.bin",
        &seal,
        "actual admitted RequestStreamControlInput",
    );
    capture_live_snapshot(capture, "ss5", &mut session);
    capture.rows(
        "live-stream-state.bin",
        &session.request_streams.checkpoint(),
        "actual RequestStreamsCheckpoint including domain/cursor/sealed outcomes",
    );
}

fn capture_reader_probes(capture: &mut Capture) {
    for legacy in [true, false] {
        let directory = tempfile::tempdir().unwrap();
        let mut session = Session::open(
            directory.path(),
            identity(),
            config(),
            SessionLimits::default(),
        )
        .unwrap();
        elect(&mut session);
        session.submit_local(&epoch(1)).unwrap();
        let input = register(&session, 801, false);
        let mut bytes = if legacy {
            capture.envelope(
                "reader-cu1-canonical.bin",
                LEGACY_CURSOR_MAGIC,
                &LegacyCursorEnvelope {
                    schema: 1,
                    ledger: identity(),
                    domain_sequence: session.sequence(),
                    trusted_control: false,
                    input,
                },
            )
        } else {
            capture.envelope(
                "reader-cu2-canonical.bin",
                CURSOR_MAGIC,
                &CursorEnvelope {
                    schema: 2,
                    ledger: identity(),
                    domain_sequence: session.sequence(),
                    replay_floor: session.stream_bounds().floor,
                    trusted_control: false,
                    input,
                },
            )
        };
        bytes.extend_from_slice(&[0xff, 0x80, 0x00]);
        let stem = if legacy { "reader-cu1" } else { "reader-cu2" };
        capture.bytes(
            &format!("{stem}-trailing.bin"),
            &bytes,
            "crafted historical suffix probe, not writer output",
        );
        let (receipt, _charge) = session.apply_cursor_entry(&bytes, 999).unwrap();
        capture.rows(
            &format!("{stem}-accepted.bin"),
            &receipt,
            "actual original apply_cursor_entry accepted trailing payload",
        );
    }
    let directory = tempfile::tempdir().unwrap();
    let mut session = Session::open(
        directory.path(),
        identity(),
        config(),
        SessionLimits::default(),
    )
    .unwrap();
    elect(&mut session);
    session.submit_local(&epoch(1)).unwrap();
    let input = register(&session, 802, false);
    session.submit_cursor_local(&input).unwrap();
    let envelope = MaintenanceEnvelope {
        schema: 1,
        ledger: identity(),
        domain_sequence: session.sequence(),
        replay_floor: session.stream_bounds().floor,
        command: CursorCommand {
            expected_revision: session.cursor_revision(),
            now: 1000,
            operation: CursorOperation::AdvanceFloor {
                through: session.cursors.checkpoint().floor,
            },
        },
    };
    let mut bytes = capture.envelope(
        "reader-cm1-canonical.bin",
        CURSOR_MAINTENANCE_MAGIC,
        &envelope,
    );
    bytes.extend_from_slice(&[0xff, 0x80, 0x00]);
    capture.bytes(
        "reader-cm1-trailing.bin",
        &bytes,
        "crafted historical suffix probe, not writer output",
    );
    session.apply_maintenance_entry(&bytes).unwrap();
    capture.rows(
        "reader-cm1-accepted.bin",
        session.cursors.checkpoint(),
        "actual original maintenance accepted trailing payload",
    );
    // All Session snapshot readers already reject an otherwise valid body's tail.
    for version in 1..=5 {
        let canonical = std::fs::read(capture.root.join(format!("live-ss{version}.bin"))).unwrap();
        let context_name = if version < 3 {
            "live-ss3.context".to_owned()
        } else {
            format!("live-ss{version}.context")
        };
        let (index, term, configuration): (u64, u64, MembershipConfiguration) =
            postcard::from_bytes(&std::fs::read(capture.root.join(context_name)).unwrap()).unwrap();
        let directory = tempfile::tempdir().unwrap();
        let mut fresh = Session::open(
            directory.path(),
            identity(),
            config(),
            SessionLimits::default(),
        )
        .unwrap();
        if version == 5 {
            elect(&mut fresh);
            durable_managed_support(&mut fresh);
        }
        fresh
            .restore_snapshot(&canonical, index, term, &configuration)
            .unwrap();
        let before = fresh.core.encode_checkpoint().unwrap();
        let mut suffixed = canonical;
        suffixed.extend_from_slice(&[0xff, 0x80, 0x00]);
        assert!(matches!(
            fresh.restore_snapshot(&suffixed, index, term, &configuration),
            Err(LedgerError::Corrupt)
        ));
        assert_eq!(fresh.core.encode_checkpoint().unwrap(), before);
        capture.bytes(
            &format!("reader-ss{version}-trailing.bin"),
            &suffixed,
            "actual original restore_snapshot rejects suffix without core publication",
        );
    }
}
